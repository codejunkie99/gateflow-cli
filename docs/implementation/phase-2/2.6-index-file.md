## Usage After Index File Created

### In SVIndexerAdapter (Module 2.7)

```typescript
import {
    extractFromIndex,
    createExtractionOptions,
    formatExtractionSummary
} from '../memory/extractors/index.js';

// After indexing
const result = await extractFromIndex(
    this.project,
    knowledgeStore,
    createExtractionOptions(knowledgeStore.getProjectId())
);

this.bus.emit({
    type: 'tool_result',
    tool: 'knowledge_extraction',
    ok: result.success,
    summary: formatExtractionSummary(result)
});
```

### In WatchManager Callback (Module 3.1)

```typescript
import { extractModulesOnly, createExtractionOptions } from '../memory/extractors/index.js';

// Quick extraction for single file update
const result = await extractModulesOnly(
    partialProject,
    store,
    createExtractionOptions(store.getProjectId())
);
```
