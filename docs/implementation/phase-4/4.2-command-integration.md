| Attribute | Value |

## Purpose

Call `fixLoop.persistAttemptMemory()` after the fix loop completes so that:
1. Fix patterns are saved to the knowledge store
2. Patterns are available for future sessions
3. Users see feedback about persisted patterns

```typescript
export async function fixCommand(
    ctx: CommandContext,
    file: string
): Promise<ExitCode> {
    const toolContext = buildToolContext(ctx);
    const agent = new GateFlowAgent(ctx.bus, toolContext);
    const fixLoop = new FixLoop(ctx.bus, ctx.verilator, agent, ctx.tools.file, {
        requireApproval: !ctx.options.yes
    });

    const result = await fixLoop.run(file);

    // ═══════════════════════════════════════════════════════════════════
    // NEW: Persist fix attempt patterns to knowledge store
    // ═══════════════════════════════════════════════════════════════════
    await persistFixPatterns(ctx, fixLoop);

    if (result.success) {
        return ExitCode.Success;
    } else {
        return ExitCode.Error;
    }
}

/**
 * Persist fix attempt patterns from the fix loop to the knowledge store
 * @private
 */
async function persistFixPatterns(
    ctx: CommandContext,
    fixLoop: FixLoop
): Promise<void> {
    // Get global knowledge store
    const knowledgeStore = getKnowledgeStore();
    if (!knowledgeStore) {
        return;  // No store available
    }

    try {
        const persisted = await fixLoop.persistAttemptMemory(knowledgeStore);

        if (persisted > 0) {
            ctx.bus.emit({
                type: 'status',
                phase: 'memory',
                label: `Saved ${persisted} fix pattern${persisted > 1 ? 's' : ''} for future reference`
            });

            // Also emit tool_result for structured output
            ctx.bus.emit({
                type: 'tool_result',
                tool: 'fix_pattern_persist',
                ok: true,
                summary: `Persisted ${persisted} fix patterns to knowledge store`
            });
        }

    } catch (error) {
        // Non-fatal - log and continue
        console.warn(
            '[fixCommand] Failed to persist fix patterns:',
            error instanceof Error ? error.message : error
        );
    }
}
```

### Step 3: Optional - Load Historical Patterns Before Fix Loop

```typescript
export async function fixCommand(
    ctx: CommandContext,
    file: string
): Promise<ExitCode> {
    const toolContext = buildToolContext(ctx);
    const agent = new GateFlowAgent(ctx.bus, toolContext);
    const fixLoop = new FixLoop(ctx.bus, ctx.verilator, agent, ctx.tools.file, {
        requireApproval: !ctx.options.yes
    });

    // ═══════════════════════════════════════════════════════════════════
    // NEW (Optional): Load historical patterns before starting
    // ═══════════════════════════════════════════════════════════════════
    await loadHistoricalFixPatterns(ctx, fixLoop, file);

    const result = await fixLoop.run(file);

    // Persist patterns
    await persistFixPatterns(ctx, fixLoop);

    if (result.success) {
        return ExitCode.Success;
    } else {
        return ExitCode.Error;
    }
}

/**
 * Load historical fix patterns relevant to the file being fixed
 * @private
 */
async function loadHistoricalFixPatterns(
    ctx: CommandContext,
    fixLoop: FixLoop,
    filePath: string
): Promise<void> {
    const knowledgeStore = getKnowledgeStore();
    if (!knowledgeStore) {
        return;
    }

    try {
        const loaded = await fixLoop.loadHistoricalPatterns(knowledgeStore, filePath);

        if (loaded > 0) {
            ctx.bus.emit({
                type: 'status',
                phase: 'memory',
                label: `Loaded ${loaded} historical fix pattern${loaded > 1 ? 's' : ''}`
            });
        }
    } catch (error) {
        // Non-fatal
        console.warn('[fixCommand] Failed to load historical patterns:', error);
    }
}
```

## Event Flow

After integration:

```
User runs: gateflow fix uart.sv

1. loadHistoricalFixPatterns() (optional)
   → emit: status { phase: 'memory', label: 'Loaded 3 historical fix patterns' }

2. fixLoop.run() executes
   → emit: status { phase: 'fixing', label: 'Attempting to fix uart.sv' }
   → (various fix attempts)
   → emit: tool_result { tool: 'fix_loop', ok: true/false }

3. persistFixPatterns()
   → emit: status { phase: 'memory', label: 'Saved 2 fix patterns' }
   → emit: tool_result { tool: 'fix_pattern_persist', ok: true }

4. Return exit code
```
